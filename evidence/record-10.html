"""
研究計画書E案 §7.1 / §7.3 — 部材構造適合スコア (CSC)。

CSC(p_i) = max_{k ∈ K} ( w_k^T f_k(p_i) )

K と f_k は §7.3 のテーブルに準拠:
    long         : elongation, linearity                -> molding
    column       : verticality, linearity               -> column
    planar-edge  : planarity, boundary_continuity (rev) -> door/window 境界
    frame        : linearity, boundary_continuity       -> door/window 枠
    curved       : scattering, normal variation         -> arch

normal variation は (1 - boundary_continuity) で近似する
（法線が近傍で大きく変化する点ほど高くなる）。

重み w_k は初期値として手動設定。§12 批判2 への対応として、後段で
train上グリッドサーチによる最適化が可能だが、まずは初期値で実験を回す。
"""

from __future__ import annotations

import numpy as np


# §7.3 に対応する初期重み。各構造ごとに使う記述子と重みを定義。
DEFAULT_WEIGHTS: dict[str, dict[str, float]] = {
    "long":         {"elongation": 0.5, "linearity": 0.5},
    "column":       {"verticality": 0.5, "linearity": 0.5},
    "planar_edge":  {"planarity": 0.5, "boundary_discontinuity": 0.5},
    "frame":        {"linearity": 0.5, "boundary_continuity": 0.5},
    "curved":       {"scattering": 0.5, "normal_variation": 0.5},
}


def _build_feature(desc: dict[str, np.ndarray], name: str) -> np.ndarray:
    """記述子辞書から名前指定で取り出す。派生量は内部で構築。"""
    if name in desc:
        return desc[name]
    if name == "boundary_discontinuity":
        return 1.0 - desc["boundary_continuity"]
    if name == "normal_variation":
        return 1.0 - desc["boundary_continuity"]
    raise KeyError(f"Unknown descriptor: {name}")


def csc_score(descriptors: dict[str, np.ndarray],
              weights: dict[str, dict[str, float]] | None = None,
              structure_scales: dict[str, float] | None = None) -> np.ndarray:
    """各点に対する CSC スコアを計算。

    Parameters
    ----------
    descriptors : src.descriptors.compute_descriptors() の戻り値
    weights : 構造別の記述子重み辞書。None なら DEFAULT_WEIGHTS を使う。
    structure_scales : 各構造のスコアに掛ける倍率辞書。グリッドサーチ用。
                       None なら全構造 1.0。

    Returns
    -------
    (N,) のCSCスコア配列。
    """
    if weights is None:
        weights = DEFAULT_WEIGHTS

    N = len(next(iter(descriptors.values())))
    per_struct = np.zeros((len(weights), N), dtype=np.float64)

    for s_i, (struct_name, w_map) in enumerate(weights.items()):
        score_k = np.zeros(N, dtype=np.float64)
        for feat_name, w in w_map.items():
            score_k += w * _build_feature(descriptors, feat_name)
        scale = 1.0 if structure_scales is None else structure_scales.get(struct_name, 1.0)
        per_struct[s_i] = score_k * scale

    # max over K
    return per_struct.max(axis=0)


def geometry_score(descriptors: dict[str, np.ndarray]) -> np.ndarray:
    """§7.2 の s_geo: 曲率・エッジ等の標準的局所幾何スコア。

    ここでは curvature 相当として (1 - planarity) と
    boundary_discontinuity を平均したものを使う。
    具体的な選択は実装の自由度内。
    """
    plan = descriptors["planarity"]
    bdc = 1.0 - descriptors["boundary_continuity"]
    return 0.5 * (1.0 - plan) + 0.5 * bdc


def retention_priority(descriptors: dict[str, np.ndarray],
                       alpha: float = 0.5,
                       weights: dict[str, dict[str, float]] | None = None,
                       structure_scales: dict[str, float] | None = None,
                       ) -> np.ndarray:
    """§7.2 の s_E = alpha * s_geo + beta * CSC (beta = 1 - alpha)。"""
    if not 0.0 <= alpha <= 1.0:
        raise ValueError("alpha must be in [0, 1]")
    beta = 1.0 - alpha
    s_geo = geometry_score(descriptors)
    s_csc = csc_score(descriptors, weights=weights, structure_scales=structure_scales)
    # 0-1 正規化（各々の min-max で割り当て公平にする）
    def _norm(x):
        lo, hi = x.min(), x.max()
        return (x - lo) / (hi - lo + 1e-12)
    return alpha * _norm(s_geo) + beta * _norm(s_csc)
