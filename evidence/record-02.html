"""
計画書E案 §12 批判2 への対応 — CSC 構造重みの train 上グリッドサーチ。

設計:
- 重み w_k の各記述子内係数は default (0.5/0.5) で固定
- 各構造 k の「スコア倍率」 scale_k ∈ {0.0, 0.5, 1.0, 1.5, 2.0} を探索
- 5 構造 × 5 倍率 = 5^5 = 3125 通り → coordinate descent 2 パスで近似
- 目的関数: train タイル上の **平均 MCR** 最大化（少数クラス保持率）
- 評価: val + test で MCR / BPR / ESR を報告、漏洩なし

出力:
- results/csc_grid/best_weights.json
- results/csc_grid/eval.csv
- results/csc_grid/summary.md
"""

from __future__ import annotations

import argparse
import csv
import json
import sys
import time
from copy import deepcopy
from pathlib import Path

import numpy as np
from scipy import stats

ROOT = Path(__file__).resolve().parents[1]
sys.path.insert(0, str(ROOT))

from src.descriptors import compute_descriptors
from src.csc import DEFAULT_WEIGHTS, retention_priority
from src.metrics import mcr, bpr, esr
from src.reductions import csc_voxel, uniform_voxel, _voxel_size_for_ratio
from src.tiles import split_xy_tiles


DATA_DIRS = {
    "train": ROOT / "data" / "arch-processed" / "train",
    "val":   ROOT / "data" / "arch-processed" / "val",
    "test":  ROOT / "data" / "arch-processed" / "test",
}
TARGET_RATIO = 0.70
BOUNDARY_RADIUS = 0.05
TAU_E = 0.85
K_NEIGH = 30
N_PER_AXIS = 3
TILE_CAP_POINTS = 80_000
MIN_POINTS_PER_TILE = 5_000

STRUCTURES = list(DEFAULT_WEIGHTS.keys())   # ['long','column','planar_edge','frame','curved']
SCALE_GRID = (0.0, 0.5, 1.0, 1.5, 2.0)
N_PASSES = 2


def _load_tiles_with_descriptors(split: str) -> list[dict]:
    """split 内の全シーンを読み込み、タイル分割 + 記述子計算。"""
    out: list[dict] = []
    data_dir = DATA_DIRS[split]
    if not data_dir.exists():
        return out
    rng = np.random.default_rng(0)
    for scene_path in sorted(data_dir.iterdir()):
        if scene_path.suffix != ".npy":
            continue
        scene_id = scene_path.stem
        P = np.load(scene_path)
        tiles = split_xy_tiles(P, n_per_axis=N_PER_AXIS, min_points=MIN_POINTS_PER_TILE)
        for t_idx, tile in enumerate(tiles):
            if len(tile) > TILE_CAP_POINTS:
                idx = rng.choice(len(tile), size=TILE_CAP_POINTS, replace=False)
                idx.sort()
                tile = tile[idx]
            print(f"  computing descriptors: {split}/{scene_id} tile {t_idx} (n={len(tile)})",
                  flush=True)
            desc = compute_descriptors(tile, k=K_NEIGH)
            vox = _voxel_size_for_ratio(tile[:, :3].astype(np.float64), TARGET_RATIO)
            out.append({
                "split": split, "scene": scene_id, "tile_idx": t_idx,
                "P": tile, "desc": desc, "voxel": vox,
                "elong": desc["elongation"],
            })
    return out


def _evaluate_mcr(tiles: list[dict], scales: dict[str, float]) -> float:
    """与えた構造倍率での平均 MCR (NaN を除く) を返す。"""
    vals = []
    for t in tiles:
        prio = retention_priority(t["desc"], alpha=0.5,
                                  structure_scales=scales)
        P_red = csc_voxel(t["P"], TARGET_RATIO, priority=prio, voxel_size=t["voxel"])
        m = mcr(t["P"], P_red)
        if not np.isnan(m):
            vals.append(m)
    return float(np.mean(vals)) if vals else float("nan")


def _coordinate_descent(train_tiles: list[dict]) -> tuple[dict[str, float], list[dict]]:
    """各構造の scale を順に最適化。passes 回繰り返す。"""
    scales = {s: 1.0 for s in STRUCTURES}
    best_mcr = _evaluate_mcr(train_tiles, scales)
    trace = [{"pass": 0, "structure": "init", "scale": None,
              "scales_snapshot": dict(scales), "train_mcr": best_mcr}]
    print(f"\n[grid] init: train MCR = {best_mcr:.4f}", flush=True)

    for p in range(N_PASSES):
        for s in STRUCTURES:
            best_local = scales[s]
            best_local_mcr = best_mcr
            for t in SCALE_GRID:
                trial = dict(scales)
                trial[s] = t
                m = _evaluate_mcr(train_tiles, trial)
                trace.append({"pass": p + 1, "structure": s, "scale": t,
                              "scales_snapshot": dict(trial),
                              "train_mcr": m})
                if m > best_local_mcr:
                    best_local_mcr = m
                    best_local = t
            scales[s] = best_local
            best_mcr = best_local_mcr
            print(f"[grid] pass {p+1} struct={s}: best scale={best_local} "
                  f"→ train MCR={best_mcr:.4f}", flush=True)

    return scales, trace


def _evaluate_full(tiles: list[dict], scales: dict[str, float]) -> list[dict]:
    """val/test タイルで MCR / BPR / ESR を計測（uniform と csc 両方）。"""
    rows: list[dict] = []
    rng = np.random.default_rng(0)
    for t in tiles:
        P = t["P"]
        # uniform
        P_u = uniform_voxel(P, TARGET_RATIO, voxel_size=t["voxel"], rng=rng)
        # csc with optimized scales
        prio = retention_priority(t["desc"], alpha=0.5, structure_scales=scales)
        P_c = csc_voxel(P, TARGET_RATIO, priority=prio, voxel_size=t["voxel"])
        for name, P_red in [("uniform_voxel", P_u), ("csc_voxel_grid", P_c)]:
            rows.append({
                "split": t["split"], "scene": t["scene"], "tile_idx": t["tile_idx"],
                "method": name,
                "n_original": len(P), "n_reduced": len(P_red),
                "MCR": mcr(P, P_red),
                "BPR": bpr(P, P_red, radius=BOUNDARY_RADIUS),
                "ESR": esr(P, P_red, tau_e=TAU_E, k=K_NEIGH, elong_cached=t["elong"]),
            })
    return rows


def _paired(rows: list[dict], split: str, metric: str
            ) -> tuple[np.ndarray, np.ndarray]:
    keys = sorted({(r["scene"], r["tile_idx"]) for r in rows if r["split"] == split})
    a, b = [], []
    for k in keys:
        u = next((r for r in rows
                 if r["split"] == split and (r["scene"], r["tile_idx"]) == k
                 and r["method"] == "uniform_voxel"), None)
        c = next((r for r in rows
                 if r["split"] == split and (r["scene"], r["tile_idx"]) == k
                 and r["method"] == "csc_voxel_grid"), None)
        if not u or not c:
            continue
        if np.isnan(u[metric]) or np.isnan(c[metric]):
            continue
        a.append(u[metric])
        b.append(c[metric])
    return np.array(a), np.array(b)


def _wilcoxon_block(rows: list[dict], split: str) -> str:
    lines = [f"\n#### {split}: csc_voxel_grid vs uniform_voxel (片側 greater)\n"]
    lines.append("| 指標 | n | unif mean | csc mean | Δ | p | |\n|---|---:|---:|---:|---:|---:|:--:|\n")
    for metric in ("MCR", "BPR", "ESR"):
        a, b = _paired(rows, split, metric)
        if len(a) < 5:
            lines.append(f"| {metric} | {len(a)} | — | — | — | (too few) | — |\n")
            continue
        try:
            res = stats.wilcoxon(b, a, alternative="greater", zero_method="wilcox")
            p = float(res.pvalue)
        except ValueError as e:
            lines.append(f"| {metric} | {len(a)} | {a.mean():.4f} | {b.mean():.4f} | "
                         f"{(b-a).mean():+.4f} | err: {e} | — |\n")
            continue
        sig = "✅" if p < 0.05 / 3 else "—"
        lines.append(
            f"| {metric} | {len(a)} | {a.mean():.4f} | {b.mean():.4f} | "
            f"{(b-a).mean():+.4f} | {p:.4g} | {sig} |\n"
        )
    return "".join(lines)


def main(quick: bool = False) -> None:
    if quick:
        global N_PASSES, SCALE_GRID
        N_PASSES = 1
        SCALE_GRID = (0.5, 1.0, 2.0)

    out_dir = ROOT / "results" / "csc_grid"
    out_dir.mkdir(parents=True, exist_ok=True)

    print("=== Loading train tiles + computing descriptors ===")
    t0 = time.time()
    train_tiles = _load_tiles_with_descriptors("train")
    print(f"loaded {len(train_tiles)} train tiles in {time.time()-t0:.1f}s")

    print("\n=== Loading val + test tiles ===")
    t0 = time.time()
    val_tiles = _load_tiles_with_descriptors("val")
    test_tiles = _load_tiles_with_descriptors("test")
    print(f"loaded {len(val_tiles)} val + {len(test_tiles)} test tiles "
          f"in {time.time()-t0:.1f}s")

    print("\n=== Coordinate descent grid search on train ===")
    t0 = time.time()
    best_scales, trace = _coordinate_descent(train_tiles)
    print(f"\ngrid search done in {time.time()-t0:.1f}s")
    print("best scales:", best_scales)

    # Save best weights + trace
    (out_dir / "best_weights.json").write_text(
        json.dumps({
            "structure_scales": best_scales,
            "default_weights": DEFAULT_WEIGHTS,
            "alpha": 0.5,
            "trace_n": len(trace),
        }, indent=2, ensure_ascii=False)
    )

    print("\n=== Evaluating on val + test ===")
    eval_rows = _evaluate_full(val_tiles + test_tiles, best_scales)
    # Train evaluation for completeness
    eval_rows += _evaluate_full(train_tiles, best_scales)

    # Write eval.csv
    field = ["split", "scene", "tile_idx", "method",
             "n_original", "n_reduced", "MCR", "BPR", "ESR"]
    with open(out_dir / "eval.csv", "w", newline="") as f:
        writer = csv.DictWriter(f, fieldnames=field)
        writer.writeheader()
        writer.writerows(eval_rows)

    # Write summary.md
    summary = ["# CSC 重みグリッドサーチ結果\n\n"]
    summary.append(f"**設定**: coordinate descent {N_PASSES} pass, "
                   f"scale grid={SCALE_GRID}, train 上で MCR 最大化\n\n")
    summary.append(f"**最良 structure scales**:\n```json\n"
                   f"{json.dumps(best_scales, indent=2)}\n```\n")
    for split in ("train", "val", "test"):
        summary.append(_wilcoxon_block(eval_rows, split))
    (out_dir / "summary.md").write_text("".join(summary))

    print("\n=== DONE ===")
    print(f"best_weights.json: {out_dir / 'best_weights.json'}")
    print(f"eval.csv         : {out_dir / 'eval.csv'} ({len(eval_rows)} rows)")
    print(f"summary.md       : {out_dir / 'summary.md'}")
    print("\n" + (out_dir / "summary.md").read_text())


if __name__ == "__main__":
    parser = argparse.ArgumentParser()
    parser.add_argument("--quick", action="store_true",
                        help="3 scales × 1 pass で動作確認")
    args = parser.parse_args()
    main(quick=args.quick)
